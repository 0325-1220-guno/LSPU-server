Firestore security rules starter (review and deploy in Firebase Console)

Role checks read the users collection. Assign every account role and provision its profile manually in Firestore; registration never assigns roles. Test these rules in the Rules Playground before publishing.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // User profiles and roles are provisioned by trusted administrators.
    function signedIn() {
      return request.auth != null;
    }
    function verifiedUser() {
      return signedIn() && request.auth.token.email_verified == true;
    }
    function hasUserProfile() {
      return verifiedUser()
        && exists(/databases/$(database)/documents/users/$(request.auth.uid));
    }
    function role() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role;
    }
    function staff() {
      return hasUserProfile() && role() in ['admin', 'registrar'];
    }
    function faculty() {
      return hasUserProfile() && role() == 'faculty';
    }
    function isStudent(uid) {
      return hasUserProfile() && request.auth.uid == uid && role() == 'student';
    }
    function validProfile(uid) {
      return request.resource.data.keys().hasAll(['uid', 'email', 'displayName', 'role', 'createdAt'])
        && request.resource.data.keys().hasOnly(['uid', 'email', 'displayName', 'role', 'createdAt'])
        && request.resource.data.uid == uid
        && request.resource.data.email is string
        && request.resource.data.email == request.auth.token.email
        && request.resource.data.displayName is string
        && request.resource.data.displayName.size() <= 100
        && request.resource.data.role in ['student', 'faculty', 'registrar', 'admin']
        && request.resource.data.createdAt is timestamp;
    }
    function facultyOwnsClass(subjectId) {
      return faculty()
        && exists(/databases/$(database)/documents/subjects/$(subjectId))
        && get(/databases/$(database)/documents/subjects/$(subjectId)).data.facultyUid == request.auth.uid;
    }
    function validStudentRecord(uid) {
      return request.resource.data.keys().hasAll(['uid', 'fullName', 'email', 'studentNumber', 'program', 'yearLevel', 'status', 'createdAt'])
        && request.resource.data.keys().hasOnly(['uid', 'fullName', 'email', 'studentNumber', 'program', 'yearLevel', 'status', 'createdAt', 'reviewedBy', 'reviewedAt'])
        && request.resource.data.uid == uid
        && request.resource.data.fullName is string
        && request.resource.data.fullName.size() <= 100
        && request.resource.data.email is string
        && request.resource.data.studentNumber is string
        && request.resource.data.program is string
        && request.resource.data.yearLevel is string
        && request.resource.data.status in ['pending', 'active']
        && request.resource.data.email == get(/databases/$(database)/documents/users/$(uid)).data.email
        && request.resource.data.createdAt is timestamp;
    }
    function validSubject() {
      return request.resource.data.keys().hasAll(['code', 'title', 'units', 'section', 'campus', 'term', 'facultyUid', 'createdBy', 'createdAt'])
        && request.resource.data.keys().hasOnly(['code', 'title', 'units', 'section', 'campus', 'term', 'facultyUid', 'createdBy', 'createdAt'])
        && request.resource.data.code is string
        && request.resource.data.code.size() > 0
        && request.resource.data.code.size() <= 20
        && request.resource.data.title is string
        && request.resource.data.title.size() > 0
        && request.resource.data.title.size() <= 100
        && request.resource.data.units is int
        && request.resource.data.units >= 1
        && request.resource.data.units <= 12
        && request.resource.data.section is string
        && request.resource.data.section.size() > 0
        && request.resource.data.section.size() <= 30
        && request.resource.data.campus is string
        && request.resource.data.campus.size() > 0
        && request.resource.data.term is string
        && request.resource.data.term.size() > 0
        && request.resource.data.term.size() <= 40
        && request.resource.data.facultyUid is string
        && (request.resource.data.facultyUid == ''
          || (exists(/databases/$(database)/documents/users/$(request.resource.data.facultyUid))
            && get(/databases/$(database)/documents/users/$(request.resource.data.facultyUid)).data.role == 'faculty'))
        && request.resource.data.createdBy == request.auth.uid
        && request.resource.data.createdAt is timestamp;
    }
    function validSection() {
      return request.resource.data.keys().hasAll(['subjectId', 'section', 'term', 'facultyUid', 'createdAt'])
        && request.resource.data.keys().hasOnly(['subjectId', 'section', 'term', 'facultyUid', 'createdAt'])
        && request.resource.data.subjectId is string
        && request.resource.data.section is string
        && request.resource.data.section.size() > 0
        && request.resource.data.section.size() <= 30
        && request.resource.data.term is string
        && request.resource.data.term.size() > 0
        && request.resource.data.term.size() <= 40
        && request.resource.data.facultyUid is string
        && (request.resource.data.facultyUid == ''
          || (exists(/databases/$(database)/documents/users/$(request.resource.data.facultyUid))
            && get(/databases/$(database)/documents/users/$(request.resource.data.facultyUid)).data.role == 'faculty'))
        && request.resource.data.createdAt is timestamp;
    }
    function validEnrollment() {
      return request.resource.data.keys().hasAll(['studentId', 'studentUid', 'studentName', 'subjectId', 'subjectCode', 'subjectTitle', 'section', 'facultyUid', 'term', 'status', 'createdAt'])
        && request.resource.data.keys().hasOnly(['studentId', 'studentUid', 'studentName', 'subjectId', 'subjectCode', 'subjectTitle', 'section', 'facultyUid', 'term', 'status', 'createdAt'])
        && request.resource.data.studentId == request.auth.uid
        && request.resource.data.studentUid == request.auth.uid
        && request.resource.data.studentName is string
        && request.resource.data.studentName.size() <= 100
        && exists(/databases/$(database)/documents/students/$(request.auth.uid))
        && request.resource.data.studentName == get(/databases/$(database)/documents/students/$(request.auth.uid)).data.fullName
        && request.resource.data.subjectId is string
        && request.resource.data.subjectId.size() > 0
        && request.resource.data.subjectId.size() <= 128
        && request.resource.data.subjectCode is string
        && request.resource.data.subjectCode.size() <= 20
        && request.resource.data.subjectTitle is string
        && request.resource.data.subjectTitle.size() <= 100
        && request.resource.data.section is string
        && request.resource.data.section.size() <= 30
        && request.resource.data.facultyUid is string
        && request.resource.data.term is string
        && request.resource.data.term.size() <= 40
        && request.resource.data.status == 'pending'
        && request.resource.data.createdAt is timestamp
        && exists(/databases/$(database)/documents/subjects/$(request.resource.data.subjectId))
        && get(/databases/$(database)/documents/subjects/$(request.resource.data.subjectId)).data.code == request.resource.data.subjectCode
        && get(/databases/$(database)/documents/subjects/$(request.resource.data.subjectId)).data.title == request.resource.data.subjectTitle
        && get(/databases/$(database)/documents/subjects/$(request.resource.data.subjectId)).data.section == request.resource.data.section
        && get(/databases/$(database)/documents/subjects/$(request.resource.data.subjectId)).data.term == request.resource.data.term
        && get(/databases/$(database)/documents/subjects/$(request.resource.data.subjectId)).data.facultyUid == request.resource.data.facultyUid;
    }

    match /users/{uid} {
      allow read: if verifiedUser() && request.auth.uid == uid;
      // An account owner can update basic profile fields, never their UID or role.
      allow update: if verifiedUser()
        && request.auth.uid == uid
        && request.resource.data.uid == resource.data.uid
        && request.resource.data.role == resource.data.role
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['email', 'displayName'])
        && validProfile(uid);
      // Create and role assignment are performed by an administrator in Firestore.
      allow create, delete: if false;
    }

    match /students/{uid} {
      allow read: if isStudent(uid) || staff();
      allow create, delete: if false;
      allow update: if staff()
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status', 'reviewedBy', 'reviewedAt'])
        && validStudentRecord(uid)
        && request.resource.data.status in ['pending', 'active']
        && request.resource.data.reviewedBy == request.auth.uid
        && request.resource.data.reviewedAt == request.time;
    }

    match /subjects/{subjectId} {
      allow read: if hasUserProfile();
      allow create: if staff() && validSubject() && request.resource.data.createdAt == request.time;
      allow update: if staff()
        && validSubject()
        && request.resource.data.createdBy == resource.data.createdBy
        && request.resource.data.createdAt == resource.data.createdAt;
      allow delete: if staff();
    }

    match /sections/{sectionId} {
      allow read: if hasUserProfile();
      allow create: if staff() && validSection() && request.resource.data.createdAt == request.time;
      allow update: if staff()
        && validSection()
        && request.resource.data.createdAt == resource.data.createdAt;
      allow delete: if staff();
    }

    match /enrollments/{enrollmentId} {
      allow create: if isStudent(request.auth.uid)
        && validEnrollment()
        && request.resource.data.createdAt == request.time;
      allow read: if hasUserProfile()
        && (resource.data.studentUid == request.auth.uid
          || resource.data.facultyUid == request.auth.uid || staff());
      // Registrar/admin approval is pending-only. Faculty may only edit grades
      // for enrollments whose assigned subject belongs to them.
      allow update: if (staff()
          && resource.data.status == 'pending'
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status', 'reviewedBy', 'reviewedAt'])
          && request.resource.data.status in ['approved', 'rejected']
          && request.resource.data.reviewedBy == request.auth.uid
          && request.resource.data.reviewedAt == request.time)
        || (facultyOwnsClass(resource.data.subjectId)
          && resource.data.facultyUid == request.auth.uid
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['grade', 'gradeUpdatedAt'])
          && request.resource.data.grade is string
          && request.resource.data.grade.size() > 0
          && request.resource.data.grade.size() <= 20
          && request.resource.data.gradeUpdatedAt == request.time);
      allow delete: if staff();
    }

    // Explicitly deny every document not granted above.
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
